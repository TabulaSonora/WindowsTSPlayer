# The host layer, as a static library the WinUI application links.
#
# No GUI toolkit below this line, and no WinRT either: this builds and tests headless, which is what
# lets the whole of M1/M2 be verified before any interface exists. The same rule holds in the Apple
# and Linux front ends, where this directory also lives.

add_library(tsgui_host STATIC
    host/ts_session.cpp
    host/ts_song_info.cpp
    host/ts_player.cpp
    host/ts_audio_device.cpp
    host/ts_i18n.cpp)

target_include_directories(tsgui_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

target_link_libraries(tsgui_host
    PUBLIC
        ts::tabulasonora
        Threads::Threads
    PRIVATE
        tsgui_warnings)

if(WIN32)
    target_link_libraries(tsgui_host PUBLIC
        # AvSetMmThreadCharacteristicsW, for the render thread and the WASAPI feeder.
        avrt
        # WASAPI is COM: CoInitializeEx, CoCreateInstance, CoTaskMemFree.
        ole32
        # KSDATAFORMAT_SUBTYPE_IEEE_FLOAT and the MMDevice/AudioClient class and interface GUIDs are
        # defined here rather than in a header, so without this every one of them is an unresolved
        # external at link time.
        ksuser
        uuid)
endif()

# The other half of TSGUI_FAST_DEBUG: the engine is compiled /O2 in Debug up in the top-level file,
# and /Od was stripped globally to let that happen, so it is handed back here. Only our own code is
# unoptimised, which is the point -- it is what anyone would actually be stepping through.
if(TSGUI_FAST_DEBUG AND MSVC)
    target_compile_options(tsgui_host PRIVATE $<$<CONFIG:Debug>:/Od>)
endif()

# ---------------------------------------------------------------------------------------------
# The seam
#
# What the .vcxproj needs in order to link, written by the build system that knows the answer.
# Generated rather than hand-maintained so there is exactly one source of truth for these paths: a
# hand-copied include list in the vcxproj would go stale the first time vcpkg or the engine moved,
# and the failure would be a confusing compile error rather than an obviously stale file.
#
# file(GENERATE) rather than configure_file() because the values wanted are generator expressions --
# TARGET_FILE and INTERFACE_INCLUDE_DIRECTORIES are not known at configure time.
#
# One file per configuration, and that is not incidental. The Visual Studio generator is
# multi-config, so TARGET_FILE resolves differently for Debug than for RelWithDebInfo; a single
# output would be "written multiple times with different content", which CMake rejects outright.
# Naming the configuration in the file is what lets the .vcxproj import the matching one, and that
# import is what keeps the CRT consistent across the seam -- a Debug application linking a
# RelWithDebInfo host layer is LNK2038 _ITERATOR_DEBUG_LEVEL mismatch, which is the single most
# likely way to break this build.
# ---------------------------------------------------------------------------------------------
file(GENERATE
    OUTPUT "${CMAKE_BINARY_DIR}/TsguiHost.$<CONFIG>.props"
    CONTENT
"<?xml version=\"1.0\" encoding=\"utf-8\"?>
<!-- Generated by src/CMakeLists.txt. Do not edit; it is rewritten on every configure. -->
<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\">
  <ItemDefinitionGroup>
    <ClCompile>
      <AdditionalIncludeDirectories>$<JOIN:$<TARGET_PROPERTY:tsgui_host,INTERFACE_INCLUDE_DIRECTORIES>,;>;$<JOIN:$<TARGET_PROPERTY:ts::tabulasonora,INTERFACE_INCLUDE_DIRECTORIES>,;>;%(AdditionalIncludeDirectories)</AdditionalIncludeDirectories>
    </ClCompile>
    <Link>
      <AdditionalDependencies>$<TARGET_FILE:tsgui_host>;$<TARGET_FILE:tabulasonora>;avrt.lib;ole32.lib;ksuser.lib;uuid.lib;%(AdditionalDependencies)</AdditionalDependencies>
    </Link>
  </ItemDefinitionGroup>
</Project>
")
